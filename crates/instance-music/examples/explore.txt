# Calm exploration loop: 32 bars, ~87 s at 88 BPM, loops seamlessly
# (the last bar's A7 resolves to the first bar's Dmaj7).
#   gmk-music render explore.txt -o explore.mid
#   gmk-music render explore.txt -o explore.wav --loop --rate 22050 --mono
#   gmk-music render explore.txt -o area.wav --loop --variant all

title "Lantern Meadow"
tempo 88
time 4/4
key D major
swing 54

track pad   inst="Pad 2 (warm)" oct=4 vol=70 pan=-20 reverb=80 gate=100
track keys  inst="Electric Piano 1" oct=3 vol=80 pan=25 reverb=60 human=6
track bass  inst="Fingered Bass" oct=2 vol=105 human=4
track lead  inst=Flute oct=5 vol=95 pan=10 reverb=70
track vib   inst=Vibraphone oct=4 vol=85 pan=-30 reverb=70
track drums drums=brush vol=90 human=8

prog home = | Dmaj7 | Bm7 | Gmaj7 | A7sus4 A7 | Dmaj7 | F#m7 | Gmaj7 | Em7 A7 |
prog away = | Gmaj7 | F#m7 | Em7 | D/F# | Gmaj7 | A/G | F#m7 B7 | Em7 A7sus4 |

# accompaniment patterns: chord degrees, restarted on every chord
pattern arp   = 1e 5 7 10 12 10 7 5        # rolling eighths, one bar
pattern arp2  = 1q 5 7 10                  # half-speed for the B section
pattern walk  = 0q. 5e 8q 5q               # 0 = slash bass (D/F# plays F#)
pattern walk2 = 0h 5q 3q
pattern hold  = *w                         # the whole chord, voice-led

# the leitmotif (also used by the variants)
pattern theme1 = F#5q. E5e D5q A4q | B4h. A4q | B4q. C#5e D5q E5q | E5h C#5h |
pattern theme2 = F#5q. E5e D5q A5q | C#6q. B5e A5h | B5q. A5e G5q F#5q | E5w |
pattern answer = B5h A5q G5q | A5h. E5q | G5q. F#5e E5q D5q | F#5h. rq | \
                 B5h D6q B5q | C#6h. A5q | A5h F#5q D#5q | E5h rh |
pattern counter = 3h 5h                    # vibraphone harmony under the B tune

grid light
  ride   x...x...x...x...
  rim    ........x.......
grid groove
  kick   x.........x.....
  snare  ....o.......x..o
  ride   x..ox..ox..ox..o
grid groovec
  crash  x...............
  kick   x.........x.....
  snare  ....o.......x..o
  ride   ...ox..ox..ox..o
grid fill
  kick   x.......x.......
  snare  ....x...o.x.x.xX
  ride   x..ox...........

section intro bars=8
  pad:   mp hold over home
  keys:  p arp over home
  bass:  p walk over home
  drums: p light*8

section A bars=8
  pad:   mp hold over home
  keys:  mp arp over home
  bass:  mf walk over home
  lead:  mf theme1 theme2
  drums: mp groovec groove*6 fill

section B bars=8
  pad:   mf hold over away
  keys:  mp arp2 over away
  bass:  mf walk2 over away
  lead:  mf answer
  vib:   p counter over away
  drums: p light*7 fill

section A2 bars=8
  pad:   mf hold over home
  keys:  mf arp over home
  bass:  f walk over home
  vib:   f theme1-12 theme2-12             # the theme an octave lower on vibraphone
  lead:  mp o5 counter over home           # flute harmony above it
  drums: mf groovec groove*6 fill

arrange |: intro A B A2 :|

# per-area variants that share the theme
variant night
  tempo 74
  transpose -2
  inst lead "Pan Flute"
  inst keys "Orchestral Harp"
  mute drums

variant cave
  tempo 80
  inst lead Ocarina
  inst keys "Music Box"
  inst pad "Pad 7 (halo)"
  vol drums 60
