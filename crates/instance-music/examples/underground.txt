# Tense underground loop: 32 bars, ~77 s at 100 BPM, E phrygian.
# An E pedal ostinato under shifting chords (F/E is the phrygian "dread"
# chord), a heartbeat kick that grows into a pounding groove, a choir line
# in the middle, and a breakdown that winds the tension back down before
# the loop restarts.
#   gmk-music render underground.txt -o underground.wav --loop --rate 22050 --mono

title "Deep Shaft"
tempo 100
time 4/4
key E phrygian

track pulse   inst="Synth Bass 2" oct=2 vol=100 gate=55 human=6
track pad     inst="Pad 8 (sweep)" oct=3 vol=72 pan=-25 reverb=100 gate=100
track strings inst="Tremolo Strings" oct=3 vol=92 pan=20 reverb=80 gate=100
track choir   inst="Choir Aahs" oct=4 vol=84 reverb=110 gate=100
track bells   inst="Tubular Bells" oct=4 vol=72 pan=35 reverb=110
track low     inst=Timpani oct=2 vol=105 reverb=60
track drums   drums=power vol=96 human=10

prog dread = | Em | F/E | Em | F/E | Em | Dm/E | C/E | F/E |
prog climb = | Am | Am/G | Fmaj7 | E7 | Am | Am/G | Fmaj7 | Esus4 E |

pattern ost   = E2e E2 B2 E2 F2 E2 B2 E2            # the pedal ostinato
pattern ost2  = E2e E2 G2 E2 F2 E2 D2 E2
pattern hold  = *w
pattern stab  = *q. *e rh                          # string stabs on the downbeat

pattern bellsA = rw | rw | rw | E5h B4h | rw | rw | rw | F5w |
pattern choirB = C5h E5h | D5h C5h | C5h A4h | B4w | C5h E5h | D5h C5h | C5h A4h | A4h G#4h |
pattern timpC  = E2q r r E2e E2 | E2q rh. |

grid heart
  kick   x..x............
grid march
  kick   x..x....x..x....
  tom6   ......x.......x.
  hat    ..x...x...x...x.
grid pound
  kick   x..x..x.x..x..x.
  snare  ....X.......X...
  tom5   ..............o.
  ohat   x.......x.......
grid poundc
  crash  x...............
  kick   x..x..x.x..x..x.
  snare  ....X.......X...
  ohat   ........x.......
grid roll
  kick   x.......x.......
  snare  ....x.x.xxxxXXXX
  tom1   ..x.............
  tom3   ......x.........

section A bars=8                    # pulse and heartbeat
  pulse: mp ost*8
  pad:   p hold over dread
  bells: mp bellsA
  drums: p heart*8

section B bars=8                    # the choir rises over a minor climb
  pulse: mf ost*8
  pad:   mp hold over climb
  strings: pp expr=50->110:8bar
  strings: hold over climb
  choir: mp choirB
  drums: mp march*8

section C bars=8                    # climax: pounding drums, string stabs, timpani
  pulse: f ost2*8
  strings: f stab over dread
  choir: mf o4 hold over dread
  low:   mf timpC*4
  drums: mf poundc pound*6 roll

section D bars=8                    # breakdown: thin out, then build back into A
  pulse: mp ost*6 ost2*2
  pad:   pp hold over dread
  bells: p rw*3 F5h E5h rw*3 B4w
  low:   pp rw*6 expr=50->127:bar (E2s)*16 E2w     # a timpani roll swelling into A
  drums: pp heart*6 rw march

arrange |: A B C D :|

variant lava
  tempo 112
  inst pad "Pad 6 (metallic)"
  inst bells "Orchestra Hit"
  vol bells 60

variant ice
  tempo 90
  transpose -2
  inst pulse "Synth Bass 1"
  inst pad "Pad 7 (halo)"
  inst bells Celesta
  shift bells 12
  mute low
